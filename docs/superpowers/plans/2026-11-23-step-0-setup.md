# Step 0: Project Setup & Shared Foundations Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Initialize the Metavayn Angular application directly in the project root, configure design system tokens and Tailwind CSS, build the GSAP/Lenis smooth scroll architecture, create reusable shared UI primitives, and construct a lazy-loaded route skeleton with a living verification sandbox.

**Architecture:** Greenfield Angular 21 application using standalone components and signals. A centralized `SmoothScrollService` operates outside Angular's zone to coordinate Lenis inertial scroll physics with GSAP `ScrollTrigger` at 60fps. Reusable primitives (`ButtonComponent`, `SectionHeadingComponent`, `CounterComponent`, `[appReveal]`, `[appTilt]`) are isolated in `src/app/shared/` with strict lifecycle cleanup and `prefers-reduced-motion` compliance.

**Tech Stack:** Angular 21, SCSS, Tailwind CSS, GSAP 3.x, Lenis, Lucide-Angular, TypeScript, Git.

**Spec:** [`docs/superpowers/specs/2026-11-23-step-0-setup-design.md`](file:///d:/Santosh/Development/metavaynhiring/docs/superpowers/specs/2026-11-23-step-0-setup-design.md)

- Initialize directly in root (`--directory .`), retaining `.agents/` and `docs/`.

- Motion safety: `prefers-reduced-motion` must bypass smooth scroll and scroll reveal animations.
- Memory hygiene: All GSAP `ScrollTrigger` instances must be explicitly destroyed in `ngOnDestroy`.

- [ ] **Step 3: Implement `SmoothScrollService`**

Create `src/app/core/services/smooth-scroll.service.ts`:
```typescript
import { Injectable, inject, NgZone, PLATFORM_ID } from '@angular/core';
import { isPlatformBrowser } from '@angular/common';
import Lenis from 'lenis';
import { gsap } from 'gsap';
import { ScrollTrigger } from 'gsap/ScrollTrigger';

@Injectable({
  providedIn: 'root'
})
export class SmoothScrollService {
  private ngZone = inject(NgZone);
  private platformId = inject(PLATFORM_ID);
  private lenis: Lenis | null = null;
  private tickerCallback: ((time: number) => void) | null = null;

  init(): void {
    if (!isPlatformBrowser(this.platformId) || this.lenis) {
      return;
    }

    gsap.registerPlugin(ScrollTrigger);

    const prefersReducedMotion = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
    if (prefersReducedMotion) {
      return;
    }

    this.ngZone.runOutsideAngular(() => {
      this.lenis = new Lenis({
        duration: 1.2,
        easing: (t) => Math.min(1, 1.001 - Math.pow(2, -10 * t)),
        orientation: 'vertical',
        smoothWheel: true,
      });

      this.lenis.on('scroll', () => {
        ScrollTrigger.update();
      });

      this.tickerCallback = (time: number) => {
        this.lenis?.raf(time * 1000);
      };

      gsap.ticker.add(this.tickerCallback);
      gsap.ticker.lagSmoothing(0);
    });
  }

  scrollTo(target: string | HTMLElement, options?: Record<string, unknown>): void {
    if (this.lenis) {
      this.lenis.scrollTo(target, options);
    } else if (typeof window !== 'undefined') {
      if (typeof target === 'string') {
        const el = document.querySelector(target);
        el?.scrollIntoView({ behavior: 'smooth' });
      } else {
        target.scrollIntoView({ behavior: 'smooth' });
      }
    }
  }

  destroy(): void {
    if (this.tickerCallback) {
      gsap.ticker.remove(this.tickerCallback);
      this.tickerCallback = null;
    }
    if (this.lenis) {
      this.lenis.destroy();
      this.lenis = null;
    }
  }
}
```

Create `src/app/shared/directives/reveal.directive.ts`:
```typescript
import { Directive, ElementRef, Input, OnInit, OnDestroy, inject, NgZone, PLATFORM_ID } from '@angular/core';
import { isPlatformBrowser } from '@angular/common';
import { gsap } from 'gsap';
import { ScrollTrigger } from 'gsap/ScrollTrigger';

@Directive({
  selector: '[appReveal]',
  standalone: true
})
export class RevealDirective implements OnInit, OnDestroy {
  private el = inject(ElementRef);
  private ngZone = inject(NgZone);
  private platformId = inject(PLATFORM_ID);

  @Input() appRevealDelay = 0;
  @Input() appRevealDuration = 0.7;
  @Input() appRevealStagger = 0;
  @Input() appRevealDirection: 'up' | 'down' | 'left' | 'right' = 'up';

  private trigger: ScrollTrigger | null = null;
  private tween: gsap.core.Tween | null = null;

  ngOnInit(): void {
    if (!isPlatformBrowser(this.platformId)) return;

    gsap.registerPlugin(ScrollTrigger);

    const prefersReducedMotion = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
    if (prefersReducedMotion) {
      this.el.nativeElement.style.opacity = '1';
      return;
    }

    this.ngZone.runOutsideAngular(() => {
      let xOffset = 0;
      let yOffset = 0;

      if (this.appRevealDirection === 'up') yOffset = 24;
      else if (this.appRevealDirection === 'down') yOffset = -24;
      else if (this.appRevealDirection === 'left') xOffset = 24;
      else if (this.appRevealDirection === 'right') xOffset = -24;

      this.tween = gsap.fromTo(
        this.el.nativeElement,
        { opacity: 0, x: xOffset, y: yOffset },
        {
          opacity: 1,
          x: 0,
          y: 0,
          duration: this.appRevealDuration,
          delay: this.appRevealDelay,
          stagger: this.appRevealStagger,
          ease: 'power3.out',
          scrollTrigger: {
            trigger: this.el.nativeElement,
            start: 'top 85%',
            once: true,
            onRefresh: (self) => {
              this.trigger = self;
            }
          }
        }
      );
    });
  }

  ngOnDestroy(): void {
    this.tween?.kill();
    this.trigger?.kill();
  }
}
```

Create `src/app/shared/directives/tilt-card.directive.ts`:
```typescript
import { Directive, ElementRef, HostListener, Input, inject, NgZone, PLATFORM_ID } from '@angular/core';
import { isPlatformBrowser } from '@angular/common';

@Directive({
  selector: '[appTilt]',
  standalone: true
})
export class TiltCardDirective {
  private el = inject(ElementRef);
  private ngZone = inject(NgZone);
  private platformId = inject(PLATFORM_ID);

  @Input() maxTilt = 8;
  @Input() scale = 1.02;

  @HostListener('mousemove', ['$event'])
  onMouseMove(e: MouseEvent): void {
    if (!isPlatformBrowser(this.platformId)) return;
    if (window.matchMedia('(prefers-reduced-motion: reduce)').matches) return;

    const rect = this.el.nativeElement.getBoundingClientRect();
    const x = e.clientX - rect.left;
    const y = e.clientY - rect.top;

    const centerX = rect.width / 2;
    const centerY = rect.height / 2;

    const rotateX = ((y - centerY) / centerY) * -this.maxTilt;
    const rotateY = ((x - centerX) / centerX) * this.maxTilt;

    this.el.nativeElement.style.transform = `perspective(1000px) rotateX(${rotateX}deg) rotateY(${rotateY}deg) scale3d(${this.scale}, ${this.scale}, ${this.scale})`;
    this.el.nativeElement.style.transition = 'transform 0.1s ease-out';
  }

  @HostListener('mouseleave')
  onMouseLeave(): void {
    if (!isPlatformBrowser(this.platformId)) return;
    this.el.nativeElement.style.transform = 'perspective(1000px) rotateX(0deg) rotateY(0deg) scale3d(1, 1, 1)';
    this.el.nativeElement.style.transition = 'transform 0.4s ease-out';
  }
}
```

```typescript
import { Component, Input, ElementRef, inject, NgZone, PLATFORM_ID, AfterViewInit, OnDestroy } from '@angular/core';
import { CommonModule, isPlatformBrowser } from '@angular/common';
import { RouterLink } from '@angular/router';
import { gsap } from 'gsap';

@Component({
  selector: 'app-button',
  standalone: true,
  imports: [CommonModule, RouterLink],
  template: `
    @if (href) {
      <a
        [routerLink]="isExternal ? null : href"
        [href]="isExternal ? href : null"
        [target]="isExternal ? '_blank' : null"
        [rel]="isExternal ? 'noopener noreferrer' : null"
        [class]="buttonClasses"
        [attr.aria-label]="ariaLabel"
      >
        <span #content class="inline-flex items-center gap-2 pointer-events-none">
          <ng-content></ng-content>
        </span>
      </a>
    } @else {
      <button
        [type]="type"
        [disabled]="disabled"
        [class]="buttonClasses"
        [attr.aria-label]="ariaLabel"
      >
        <span #content class="inline-flex items-center gap-2 pointer-events-none">
          <ng-content></ng-content>
        </span>
      </button>
    }
  `
})
export class ButtonComponent implements AfterViewInit, OnDestroy {
  private el = inject(ElementRef);
  private ngZone = inject(NgZone);
  private platformId = inject(PLATFORM_ID);

  @Input() variant: 'primary' | 'ghost' | 'amber-cta' = 'primary';
  @Input() type: 'button' | 'submit' = 'button';
  @Input() disabled = false;
  @Input() href?: string;
  @Input() isExternal = false;
  @Input() ariaLabel?: string;

  private quickToX?: (n: number) => void;
  private quickToY?: (n: number) => void;
  private mouseMoveHandler?: (e: MouseEvent) => void;
  private mouseLeaveHandler?: () => void;

  get buttonClasses(): string {
    const base = 'relative inline-flex items-center justify-center font-semibold rounded-xl px-6 py-3 transition-all duration-300 cursor-pointer disabled:opacity-50 disabled:cursor-not-allowed select-none';
    
    switch (this.variant) {
      case 'amber-cta':
        return `${base} bg-amber text-midnight font-bold glow-amber hover:brightness-110 active:scale-95`;
      case 'ghost':
        return `${base} bg-transparent text-white border border-white/20 hover:border-cyan hover:text-cyan active:scale-95`;
      case 'primary':
      default:
        return `${base} bg-royal text-white hover:bg-electric glow-cyan active:scale-95`;
    }
  }

  ngAfterViewInit(): void {
    if (!isPlatformBrowser(this.platformId)) return;
    if (window.matchMedia('(prefers-reduced-motion: reduce)').matches) return;

    this.ngZone.runOutsideAngular(() => {
      const container = this.el.nativeElement.querySelector('button, a');
      if (!container) return;

      this.quickToX = gsap.quickTo(container, 'x', { duration: 0.3, ease: 'power2.out' });
      this.quickToY = gsap.quickTo(container, 'y', { duration: 0.3, ease: 'power2.out' });

      this.mouseMoveHandler = (e: MouseEvent) => {
        const rect = container.getBoundingClientRect();
        const distCenterRelX = (e.clientX - (rect.left + rect.width / 2)) * 0.25;
        const distCenterRelY = (e.clientY - (rect.top + rect.height / 2)) * 0.25;
        this.quickToX?.(distCenterRelX);
        this.quickToY?.(distCenterRelY);
      };

      this.mouseLeaveHandler = () => {
        this.quickToX?.(0);
        this.quickToY?.(0);
      };

      container.addEventListener('mousemove', this.mouseMoveHandler);
      container.addEventListener('mouseleave', this.mouseLeaveHandler);
    });
  }

  ngOnDestroy(): void {
    const container = this.el.nativeElement.querySelector('button, a');
    if (container) {
      if (this.mouseMoveHandler) container.removeEventListener('mousemove', this.mouseMoveHandler);
      if (this.mouseLeaveHandler) container.removeEventListener('mouseleave', this.mouseLeaveHandler);
    }
  }
}
```

```typescript
import { Component, Input, ElementRef, inject, NgZone, PLATFORM_ID, AfterViewInit, OnDestroy, signal } from '@angular/core';
import { CommonModule, isPlatformBrowser } from '@angular/common';
import { gsap } from 'gsap';
import { ScrollTrigger } from 'gsap/ScrollTrigger';

@Component({
  selector: 'app-counter',
  standalone: true,
  imports: [CommonModule],
  template: `
    <span class="font-heading font-bold text-3xl md:text-5xl text-white">
      {{ prefix }}{{ displayValue() }}{{ suffix }}
    </span>
  `
})
export class CounterComponent implements AfterViewInit, OnDestroy {
  private el = inject(ElementRef);
  private ngZone = inject(NgZone);
  private platformId = inject(PLATFORM_ID);

  @Input({ required: true }) targetValue!: number;
  @Input() duration = 2;
  @Input() prefix = '';
  @Input() suffix = '';
  @Input() decimals = 0;

  displayValue = signal<string>('0');
  private tween: gsap.core.Tween | null = null;
  private trigger: ScrollTrigger | null = null;

  ngAfterViewInit(): void {
    if (!isPlatformBrowser(this.platformId)) {
      this.displayValue.set(this.targetValue.toFixed(this.decimals));
      return;
    }

    gsap.registerPlugin(ScrollTrigger);

    if (window.matchMedia('(prefers-reduced-motion: reduce)').matches) {
      this.displayValue.set(this.targetValue.toFixed(this.decimals));
      return;
    }

    this.ngZone.runOutsideAngular(() => {
      const obj = { val: 0 };
      this.tween = gsap.to(obj, {
        val: this.targetValue,
        duration: this.duration,
        ease: 'power2.out',
        scrollTrigger: {
          trigger: this.el.nativeElement,
          start: 'top 85%',
          once: true,
          onRefresh: (self) => {
            this.trigger = self;
          }
        },
        onUpdate: () => {
          this.ngZone.run(() => {
            this.displayValue.set(obj.val.toFixed(this.decimals));
          });
        }
      });
    });
  }

  ngOnDestroy(): void {
    this.tween?.kill();
    this.trigger?.kill();
  }
}
```
